//Énoncé: Écrire une fonction fibonacci(n) qui retourne le nième nombre de la suite de Fibonacci.
//Rappel:
//fib(0) = 0
//fib(1) = 1
//fib(n) = fib(n-1) + fib(n-2)

// Version récursive
function fibonacci(n) {
 if n=0 then val=0
 elseif n=1 then val=1
else val=fibonacci(n-2)+fibonacci(n-1)
return val
}
